import React from 'react'
import {createRoot} from 'react-dom/client'
import {RoundStatsPanel} from '@/components/admin/RoundStatsPanel'
import {summarizeRoundStats} from '@/lib/matching/round-stats'

const count=mode==='empty'?0:mode==='single'?18:54
const people=Array.from({length:count},(_,index)=>({
  game_type_pref:index<25?'双人':index<40?'多人':'都可以',
  gender_pref:mode==='single'?'女':index<25?'男':index<42?'女':index<52?'都可以':null,
  member:{member_identity:{gender:mode==='single'?'female':index<20?'male':index<51?'female':'other'}},
  availability:Object.fromEntries(Array.from({length:14},(_,day)=>[
    `2026-11-${String(day+4).padStart(2,'0')}`,
    ['上午','下午','晚上'].filter((_,period)=>(index+day+period)%3!==0),
  ])),
}))

  <RoundStatsPanel stats={summarizeRoundStats(people)} activityStart="2026-11-04" activityEnd="2026-11-17"/>
